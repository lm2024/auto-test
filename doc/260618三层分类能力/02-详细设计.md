## 1. 数据库设计

### 1.1 分类表 DDL

```sql
CREATE TABLE `test_category` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '主键',
  `category_name` VARCHAR(100) NOT NULL COMMENT '分类名称',
  `parent_id` BIGINT NOT NULL DEFAULT 0 COMMENT '父分类ID，0表示顶级',
  `level` INT NOT NULL DEFAULT 1 COMMENT '层级：1/2/3',
  `path` VARCHAR(500) NOT NULL DEFAULT '/' COMMENT '层级路径，如 /1/5/12/',
  `sort_order` INT NOT NULL DEFAULT 0 COMMENT '排序号',
  `description` VARCHAR(500) DEFAULT NULL COMMENT '描述',
  `status` INT NOT NULL DEFAULT 1 COMMENT '状态：0禁用 1启用',
  `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  KEY `idx_parent_id` (`parent_id`),
  KEY `idx_level` (`level`),
  KEY `idx_path` (`path`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='测试分类表';
```

#### CategoryServiceImpl.java
```java
package com.autotest.service.impl;

import com.autotest.mapper.CategoryMapper;
import com.autotest.model.dto.CategoryCreateDTO;
import com.autotest.model.entity.Category;
import com.autotest.model.vo.CategoryVO;
import com.autotest.service.CategoryService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

@Service
public class CategoryServiceImpl implements CategoryService {
    
    @Autowired
    private CategoryMapper categoryMapper;
    
    @Override
    public List<CategoryVO> getCategoryTree() {
        List<Category> allCategories = categoryMapper.selectAll();
        return buildTree(allCategories, 0L);
    }
    
    private List<CategoryVO> buildTree(List<Category> all, Long parentId) {
        return all.stream()
            .filter(c -> c.getParentId().equals(parentId))
            .map(c -> {
                CategoryVO vo = convertToVO(c);
                vo.setChildren(buildTree(all, c.getId()));
                return vo;
            })
            .collect(Collectors.toList());
    }
    
    @Override
    public List<CategoryVO> getCategoriesByLevel(Integer level) {
        return categoryMapper.selectAll().stream()
            .filter(c -> c.getLevel().equals(level))
            .map(this::convertToVO)
            .collect(Collectors.toList());
    }
    
    @Override
    public List<CategoryVO> getChildrenByParentId(Long parentId) {
        return categoryMapper.selectByParentId(parentId).stream()
            .map(this::convertToVO)
            .collect(Collectors.toList());
    }
    
    @Override
    @Transactional
    public CategoryVO createCategory(CategoryCreateDTO dto) {
        // 校验层级
        if (dto.getLevel() > 3) {
            throw new RuntimeException("最多支持3层分类");
        }
        
        // 校验父分类
        Long parentId = dto.getParentId() != null ? dto.getParentId() : 0L;
        if (parentId > 0) {
            Category parent = categoryMapper.selectById(parentId);
            if (parent == null) {
                throw new RuntimeException("父分类不存在");
            }
            if (parent.getLevel() >= 3) {
                throw new RuntimeException("父分类已达最大层级，无法继续添加子分类");
            }
        }
        
        // 构建路径
        String path = "/";
        if (parentId > 0) {
            Category parent = categoryMapper.selectById(parentId);
            path = parent.getPath();
        }
        
        // 获取排序号
        Integer maxSort = categoryMapper.getMaxSortOrder(parentId);
        Integer sortOrder = dto.getSortOrder() != null ? dto.getSortOrder() : (maxSort != null ? maxSort + 1 : 1);
        
        Category category = new Category();
        category.setCategoryName(dto.getCategoryName());
        category.setParentId(parentId);
        category.setLevel(dto.getLevel());
        category.setSortOrder(sortOrder);
        category.setDescription(dto.getDescription());
        category.setStatus(1);
        
        categoryMapper.insert(category);
        
        // 更新路径
        path = path + category.getId() + "/";
        category.setPath(path);
        categoryMapper.update(category);
        
        return convertToVO(category);
    }
    
    @Override
    @Transactional
    public CategoryVO updateCategory(Long id, CategoryCreateDTO dto) {
        Category category = categoryMapper.selectById(id);
        if (category == null) {
            throw new RuntimeException("分类不存在");
        }
        
        category.setCategoryName(dto.getCategoryName());
        if (dto.getSortOrder() != null) {
            category.setSortOrder(dto.getSortOrder());
        }
        category.setDescription(dto.getDescription());
        
        categoryMapper.update(category);
        return convertToVO(category);
    }
    
    @Override
    @Transactional
    public void deleteCategory(Long id) {
        // 检查是否有子分类
        int childCount = categoryMapper.countChildren(id);
        if (childCount > 0) {
            throw new RuntimeException("该分类下有子分类，无法删除");
        }
        
        // 检查是否有关联链路
        int chainCount = categoryMapper.countChainsByCategoryId(id);
        if (chainCount > 0) {
            throw new RuntimeException("该分类下有链路，无法删除");
        }
        
        categoryMapper.softDelete(id);
    }
    
    @Override
    public List<CategoryVO> getCategoryPath(Long categoryId) {
        List<CategoryVO> path = new ArrayList<>();
        Category current = categoryMapper.selectById(categoryId);
        while (current != null && current.getId() > 0) {
            path.add(0, convertToVO(current));
            if (current.getParentId() == 0) break;
            current = categoryMapper.selectById(current.getParentId());
        }
        return path;
    }
    
    @Override
    public List<Long> getAllChildIds(Long categoryId) {
        List<Long> ids = new ArrayList<>();
        ids.add(categoryId);
        
        Category category = categoryMapper.selectById(categoryId);
        if (category != null) {
            List<Category> children = categoryMapper.selectByPath(category.getPath());
            children.forEach(c -> ids.add(c.getId()));
        }
        
        return ids;
    }
    
    private CategoryVO convertToVO(Category category) {
        CategoryVO vo = new CategoryVO();
        vo.setId(category.getId());
        vo.setCategoryName(category.getCategoryName());
        vo.setParentId(category.getParentId());
        vo.setLevel(category.getLevel());
        vo.setPath(category.getPath());
        vo.setSortOrder(category.getSortOrder());
        vo.setDescription(category.getDescription());
        vo.setStatus(category.getStatus());
        return vo;
    }
}
```

### 2.4 Controller

#### CategoryController.java
```java
package com.autotest.controller;

import com.autotest.common.Result;
import com.autotest.model.dto.CategoryCreateDTO;
import com.autotest.model.vo.CategoryVO;
import com.autotest.service.CategoryService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import javax.validation.Valid;
import java.util.List;

@RestController
@RequestMapping("/api/category")
public class CategoryController {
    
    @Autowired
    private CategoryService categoryService;
    
    @GetMapping("/tree")
    public Result<List<CategoryVO>> getTree() {
        return Result.success(categoryService.getCategoryTree());
    }
    
    @GetMapping("/list")
    public Result<List<CategoryVO>> getList(
            @RequestParam(required = false) Integer level,
            @RequestParam(required = false) Long parentId) {
        if (level != null) {
            return Result.success(categoryService.getCategoriesByLevel(level));
        }
        if (parentId != null) {
            return Result.success(categoryService.getChildrenByParentId(parentId));
        }
        return Result.success(categoryService.getCategoryTree());
    }
    
    @GetMapping("/children")
    public Result<List<CategoryVO>> getChildren(@RequestParam Long parentId) {
        return Result.success(categoryService.getChildrenByParentId(parentId));
    }
    
    @GetMapping("/path/{id}")
    public Result<List<CategoryVO>> getCategoryPath(@PathVariable Long id) {
        return Result.success(categoryService.getCategoryPath(id));
    }
    
    @PostMapping("/create")
    public Result<CategoryVO> create(@Valid @RequestBody CategoryCreateDTO dto) {
        return Result.success(categoryService.createCategory(dto));
    }
    
    @PutMapping("/update/{id}")
    public Result<CategoryVO> update(@PathVariable Long id, @Valid @RequestBody CategoryCreateDTO dto) {
        return Result.success(categoryService.updateCategory(id, dto));
    }
    
    @DeleteMapping("/delete/{id}")
    public Result<Void> delete(@PathVariable Long id) {
        categoryService.deleteCategory(id);
        return Result.success(null);
    }
}
```

```java
@PostMapping("/batchByCategory")
public Result<?> batchExecuteByCategory(@RequestBody Map<String, Object> params) {
    Long categoryId = Long.valueOf(params.get("categoryId").toString());
    List<Long> chainIds = chainService.getChainIdsByCategoryId(categoryId);
    
    List<Map<String, Object>> results = new ArrayList<>();
    for (Long chainId : chainIds) {
        TestChain chain = chainMapper.selectById(chainId);
        if (chain != null) {
            try {
                String executionId = executeService.runChain(chain.getChainCode());
                results.add(Map.of("chainCode", chain.getChainCode(), "status", "success", "executionId", executionId));
            } catch (Exception e) {
                results.add(Map.of("chainCode", chain.getChainCode(), "status", "failed", "message", e.getMessage()));
            }
        }
    }
    return Result.success(results);
}
```

```javascript
var categoryTree = [];
var selectedCategoryId = null;

function loadCategoryTree() {
  var baseUrl = settings.platformUrl || 'http://localhost:9093';
  fetch(baseUrl + '/api/category/tree')
    .then(function(r) { return r.json(); })
    .then(function(d) {
      if (d.code === 200) {
        categoryTree = d.data || [];
        renderCategoryOptions(1, 0);
      }
    });
}

function renderCategoryOptions(level, parentId) {
  var selectId = 'pushCatLevel' + level;
  var select = el(selectId);
  if (!select) return;
  
  select.innerHTML = '<option value="">请选择' + ['', '一级', '二级', '三级'][level] + '分类</option>';
  
  var items = categoryTree;
  if (parentId > 0) {
    var parent = findCategoryById(categoryTree, parentId);
    items = parent ? (parent.children || []) : [];
  }
  
  items.forEach(function(cat) {
    var opt = document.createElement('option');
    opt.value = cat.id;
    opt.textContent = cat.categoryName;
    select.appendChild(opt);
  });
  
  select.style.display = items.length > 0 ? '' : 'none';
  
  // 隐藏后续层级
  for (var i = level + 1; i <= 3; i++) {
    var nextSelect = el('pushCatLevel' + i);
    if (nextSelect) nextSelect.style.display = 'none';
  }
}

function findCategoryById(tree, id) {
  for (var i = 0; i < tree.length; i++) {
    if (tree[i].id === id) return tree[i];
    if (tree[i].children) {
      var found = findCategoryById(tree[i].children, id);
      if (found) return found;
    }
  }
  return null;
}

// 绑定事件
el('pushCatLevel1').addEventListener('change', function() {
```
