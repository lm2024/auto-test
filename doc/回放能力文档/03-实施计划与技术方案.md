### 2.1 插件端 HTTP 回放引擎

```javascript
// replay.js - HTTP 回放模式
class HttpReplayer {
  constructor(chain, options = {}) {
    this.chain = chain;
    this.nodes = chain.nodes.sort((a, b) => a.sortNo - b.sortNo);
    this.results = [];
    this.isPaused = false;
    this.isStopped = false;
    this.onNodeComplete = options.onNodeComplete || (() => {});
    this.onLog = options.onLog || (() => {});
    this.onComplete = options.onComplete || (() => {});
  }

  async start() {
    this.onLog('info', `开始HTTP回放: ${this.chain.chainName}，共 ${this.nodes.length} 个接口`);
    
    // 分组：支持并行组
    const groups = this.buildGroups(this.nodes);
    
    for (const group of groups) {
      if (this.isStopped) break;
      
      if (group.length === 1) {
        // 串行执行
        await this.executeNode(group[0]);
      } else {
        // 并行执行
        await Promise.all(group.map(node => this.executeNode(node)));
      }
    }
    
    this.onComplete(this.results);
    return this.results;
  }

  async executeNode(node) {
    if (this.isStopped) return;
    while (this.isPaused) {
      await this.sleep(100);
      if (this.isStopped) return;
    }
    
    const startTime = Date.now();
    this.onLog('info', `[${this.getNodeIndex(node) + 1}/${this.nodes.length}] 正在执行: ${node.nodeName}`);
    
    try {
      const response = await fetch(node.requestUrl, {
        method: node.requestMethod,
        headers: JSON.parse(node.requestHeaders || '{}'),
        body: ['POST', 'PUT', 'PATCH'].includes(node.requestMethod) && node.bodyData
          ? node.bodyData
          : undefined,
        credentials: 'include' // 使用浏览器当前Cookie
      });
      
      const durationMs = Date.now() - startTime;
      const responseBody = await response.text();
      
      const result = {
        node,
        status: 'SUCCESS',
        responseCode: response.status,
        responseHeaders: Object.fromEntries(response.headers.entries()),
        responseBody,
        durationMs
      };
      
      this.results.push(result);
      this.onNodeComplete(result);
      this.onLog('info', `✓ ${node.nodeName} - ${response.status} - ${durationMs}ms`);
      
    } catch (error) {
      const durationMs = Date.now() - startTime;
      const result = {
        node,
        status: 'FAILED',
        error: error.message,
        durationMs
      };
      
      this.results.push(result);
      this.onNodeComplete(result);
      this.onLog('error', `✗ ${node.nodeName} - ${error.message} - 继续执行`);
    }
  }

  // 按并行组分组
  buildGroups(nodes) {
    const groups = [];
    let currentGroup = [];
    let currentParallelGroup = null;
    
    for (const node of nodes) {
      if (node.parallelGroup && node.parallelGroup === currentParallelGroup) {
        currentGroup.push(node);
      } else {
        if (currentGroup.length > 0) {
          groups.push(currentGroup);
        }
        currentGroup = [node];
        currentParallelGroup = node.parallelGroup || null;
      }
    }
    
    if (currentGroup.length > 0) {
      groups.push(currentGroup);
    }
    
    return groups;
  }

  pause() { this.isPaused = true; }
  resume() { this.isPaused = false; }
  stop() { this.isStopped = true; }
  sleep(ms) { return new Promise(resolve => setTimeout(resolve, ms)); }
}
```

```javascript
// replay.js - 浏览器回放模式
class BrowserReplayer {
  constructor(chain, options = {}) {
    this.chain = chain;
    this.nodes = chain.nodes;
    this.capturedApis = [];
    this.isCapturing = false;
    this.onCapture = options.onCapture || (() => {});
    this.onLog = options.onLog || (() => {});
  }

  async start() {
    this.onLog('info', `开始浏览器回放: ${this.chain.chainName}`);
    
    // 1. 打开第一个接口的URL所在页面
    const startUrl = this.extractBaseUrl(this.nodes[0].requestUrl);
    const tab = await chrome.tabs.create({ url: startUrl, active: true });
    this.tabId = tab.id;
    
    // 2. 等待页面加载
    await this.waitForTabLoad(tab.id);
    
    // 3. 激活网络捕获
    await this.startCapture(tab.id);
    
    this.onLog('info', '页面已打开，网络捕获已激活，请在页面上进行操作');
    this.onLog('info', '操作完成后点击「完成回放」按钮');
    
    // 4. 等待用户操作（由外部调用 stop() 结束）
    return new Promise((resolve) => {
      this.resolveStop = resolve;
    });
  }

  async startCapture(tabId) {
    this.isCapturing = true;
    
    // 通过 background.js 激活 CDP 捕获
    chrome.runtime.sendMessage({
      type: 'START_CAPTURE',
      tabId: tabId
    });
    
    // 监听捕获到的请求
    chrome.runtime.onMessage.addListener((message) => {
      if (message.type === 'CAPTURED_API' && message.tabId === tabId) {
        this.capturedApis.push(message.api);
        this.onCapture(message.api);
      }
    });
  }

  async stop() {
    this.isCapturing = false;
    
    // 停止捕获
    chrome.runtime.sendMessage({
      type: 'STOP_CAPTURE',
      tabId: this.tabId
    });
    
    // 匹配捕获的请求到链路
    const matchedResults = this.matchCapturedToChain(this.capturedApis);
    
    if (this.resolveStop) {
      this.resolveStop(matchedResults);
    }
    
    return matchedResults;
  }

  // 将捕获的请求匹配到链路接口
  matchCapturedToChain(captured) {
    return captured.map(req => {
      const matchedNode = this.nodes.find(n => 
        n.requestMethod === req.method &&
        this.normalizeUrl(n.requestUrl) === this.normalizeUrl(req.url)
      );
      
      return {
        ...req,
        matchedNode: matchedNode || null,
        changeType: matchedNode 
          ? this.detectChanges(matchedNode, req) 
          : 'ADDED', // 未匹配到 = 新增接口
        isSelected: true // 默认选中
      };
    });
  }

  // 检测变化
  detectChanges(originalNode, capturedReq) {
    const changes = [];
    
    if (originalNode.requestUrl !== capturedReq.url) {
      changes.push({ field: 'requestUrl', old: originalNode.requestUrl, new: capturedReq.url });
    }
    if (originalNode.bodyData !== capturedReq.body) {
      changes.push({ field: 'bodyData', old: originalNode.bodyData, new: capturedReq.body });
    }
    if (originalNode.requestHeaders !== JSON.stringify(capturedReq.headers)) {
      changes.push({ field: 'requestHeaders', old: originalNode.requestHeaders, new: capturedReq.headers });
    }
    
    return changes.length > 0 ? 'MODIFIED' : 'UNCHANGED';
  }

  extractBaseUrl(url) {
    try {
      const u = new URL(url);
      return u.origin;
    } catch {
      return url;
    }
  }

  normalizeUrl(url) {
    const q = url.indexOf('?');
    return q > 0 ? url.substring(0, q) : url;
  }
}
```

### 2.3 后端 Diff 引擎

```java
@Service
public class DiffService {
    
    @Autowired
    private NodeSnapshotMapper snapshotMapper;
    
    @Autowired
    private AiService aiService;
    
    public DiffResult computeDiff(String chainCode, int currentVersion) {
        // 获取当前版本和上一版本的节点快照
        ChainVersion currentVersionObj = versionMapper.selectByChainCodeAndVersion(chainCode, currentVersion);
        ChainVersion previousVersionObj = versionMapper.selectByChainCodeAndVersion(chainCode, currentVersion - 1);
        
        if (previousVersionObj == null) {
            // 第一个版本，所有节点都是新增
            List<NodeSnapshot> current = snapshotMapper.selectByVersionId(currentVersionObj.getId());
            DiffResult result = new DiffResult();
            current.forEach(result::addAdded);
            return result;
        }
        
        List<NodeSnapshot> current = snapshotMapper.selectByVersionId(currentVersionObj.getId());
        List<NodeSnapshot> previous = snapshotMapper.selectByVersionId(previousVersionObj.getId());
        
        return computeDiffFromSnapshots(current, previous);
    }
    
    public DiffResult computeDiffFromSnapshots(List<NodeSnapshot> current, List<NodeSnapshot> previous) {
        DiffResult result = new DiffResult();
        
        Map<String, NodeSnapshot> prevMap = previous.stream()
            .collect(Collectors.toMap(NodeSnapshot::getNodeName, Function.identity()));
        Map<String, NodeSnapshot> currMap = current.stream()
            .collect(Collectors.toMap(NodeSnapshot::getNodeName, Function.identity()));
        
        // 检测修改和删除
        for (NodeSnapshot prev : previous) {
            NodeSnapshot curr = currMap.get(prev.getNodeName());
            if (curr == null) {
                result.addRemoved(prev);
            } else {
                List<FieldChange> changes = compareFields(prev, curr);
                if (!changes.isEmpty()) {
                    result.addModified(curr, changes);
                } else {
                    result.addUnchanged(curr);
                }
            }
        }
        
        // 检测新增
        for (NodeSnapshot curr : current) {
            if (!prevMap.containsKey(curr.getNodeName())) {
                result.addAdded(curr);
            }
        }
        
        return result;
    }
    
    private List<FieldChange> compareFields(NodeSnapshot prev, NodeSnapshot curr) {
        List<FieldChange> changes = new ArrayList<>();
        checkField(changes, "requestUrl", "请求地址", prev.getRequestUrl(), curr.getRequestUrl());
        checkField(changes, "requestMethod", "请求方法", prev.getRequestMethod(), curr.getRequestMethod());
        checkField(changes, "requestHeaders", "请求头", prev.getRequestHeaders(), curr.getRequestHeaders());
        checkField(changes, "bodyData", "请求体", prev.getBodyData(), curr.getBodyData());
        checkField(changes, "responseCode", "响应码", 
            prev.getResponseCode() != null ? String.valueOf(prev.getResponseCode()) : null,
            curr.getResponseCode() != null ? String.valueOf(curr.getResponseCode()) : null);
        checkField(changes, "responseBody", "响应体", prev.getResponseBody(), curr.getResponseBody());
        return changes;
    }
    
    private void checkField(List<FieldChange> changes, String field, String label, String oldVal, String newVal) {
        if (!Objects.equals(oldVal, newVal)) {
            changes.add(new FieldChange(field, label, oldVal, newVal));
        }
    }
}
```

### 2.4 AI 差异分析

```java
@Service
public class AiService {
    
    public AiAnalysis analyzeDiff(DiffResult diff) {
        StringBuilder prompt = new StringBuilder();
        prompt.append("分析以下接口链路差异，给出影响分析和建议：\n\n");
        
        prompt.append("新增接口：\n");
        diff.getAdded().forEach(node -> 
            prompt.append(String.format("- %s %s\n", node.getRequestMethod(), node.getRequestUrl()))
        );
        
        prompt.append("\n删除接口：\n");
        diff.getRemoved().forEach(node -> 
            prompt.append(String.format("- %s %s\n", node.getRequestMethod(), node.getRequestUrl()))
        );
        
        prompt.append("\n修改接口：\n");
        diff.getModified().forEach(item -> {
            prompt.append(String.format("- %s %s\n", item.getNode().getRequestMethod(), item.getNode().getRequestUrl()));
            item.getChanges().forEach(change -> 
                prompt.append(String.format("  %s: %s → %s\n", change.getLabel(), change.getOldValue(), change.getNewValue()))
            );
        });
        
        prompt.append("\n请分析：\n1. 变更摘要\n2. 潜在影响\n3. 测试建议");
        
        // 调用AI模型
        String aiResponse = callAiModel(prompt.toString());
        
        return parseAiResponse(aiResponse);
    }
}
```

## 四、风险与应对

| 风险 | 概率 | 影响 | 应对 |
|------|------|------|------|
| 浏览器回放时页面跨域限制 | 中 | 无法捕获某些请求 | CDP 捕获不受跨域限制，fetch 拦截作为补充 |
| 并行组回放时请求顺序不确定 | 低 | 结果难以对比 | 按节点名称匹配而非顺序 |
| 大量版本查询性能 | 低 | 查询慢 | 默认只查5条，增加索引 |
| AI 服务不可用 | 中 | 无分析结果 | 降级为纯 Diff 展示，AI 为可选 |
| 回放时 Cookie 过期 | 高 | 回放失败 | 提示用户先登录，回放使用当前 Cookie |
